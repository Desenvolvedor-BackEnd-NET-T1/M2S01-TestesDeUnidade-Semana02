// Etapa 2 — Limites mínimos e máximos
//
// Como incorporar: copie este conteúdo para um novo arquivo .cs dentro de
// tests/CadastroClientes.Tests/Servicos/ (veja docs/testes-adicionais/LEIA-ME.md).
//
// Objetivo: exercitar os ramos de tamanho mínimo e máximo de cada campo que
// possui essa regra, usando dados de fronteira: o valor mínimo permitido,
// um caractere abaixo do mínimo, o valor máximo permitido e um caractere
// acima do máximo. O conteúdo das strings não importa aqui — apenas o
// comprimento — por isso elas são geradas com `new string(caractere, tamanho)`.

using CadastroClientes.Core.Modelos;
using CadastroClientes.Core.Servicos;
using CadastroClientes.Tests.TestDoubles;

namespace CadastroClientes.Tests.Servicos;

public class LimitesMinimoEMaximoTests

    // ----- Nome: 3 a 10 caracteres -----

    [Fact]
    public void Cadastrar_ComNomeNoTamanhoMinimo_DeveTerSucesso()
    {
        var repositorio = new RepositorioClienteEmMemoriaDeTeste();
        var servico = new ClienteService(repositorio);
        var dto = CadastroClienteDtoFabrica.CriarValido();
        dto.Nome = new string('a', 3);

        var resultado = servico.Cadastrar(dto);

        Assert.True(resultado.Sucesso);
    }

    [Fact]
    public void Cadastrar_ComNomeUmCaractereAbaixoDoMinimo_DeveFalhar()
    {
        var repositorio = new RepositorioClienteEmMemoriaDeTeste();
        var servico = new ClienteService(repositorio);
        var dto = CadastroClienteDtoFabrica.CriarValido();
        dto.Nome = new string('a', 2);

        var resultado = servico.Cadastrar(dto);

        Assert.False(resultado.Sucesso);
        Assert.Contains(resultado.Erros, erro => erro.Campo == nameof(Cliente.Nome));
        Assert.Empty(repositorio.Listar());
    }

    // ----- Email: até 100 caracteres (não há mínimo além da obrigatoriedade) -----

    [Fact]
    public void Cadastrar_ComEmailNoTamanhoMaximo_DeveTerSucesso()
    {
        var repositorio = new RepositorioClienteEmMemoriaDeTeste();
        var servico = new ClienteService(repositorio);
        var dto = CadastroClienteDtoFabrica.CriarValido();
        const string dominio = "@exemplo.com";
        dto.Email = new string('a', 100 - dominio.Length) + dominio;

        var resultado = servico.Cadastrar(dto);

        Assert.True(resultado.Sucesso);
        Assert.Equal(100, dto.Email.Length);
    }
